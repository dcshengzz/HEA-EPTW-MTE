CREATE PROCEDURE [dbo].[Procedure_CCPReport]
	@DateFrom	DateTime,
	@DateTo		DateTime

AS

DECLARE @TEMP table 
(
	ProjectName [varchar](150) NOT NULL,
	RegistrationNo [varchar](20) NOT NULL,
	EquipmentName [varchar](100) NULL,
	[Description] [varchar](200) NULL,
	CCP01 DateTime NULL,
	CCP02 DateTime NULL,
	CCP03 DateTime NULL,
	CCP04 DateTime NULL,
	CCP05 DateTime NULL,
	CCP06 DateTime NULL,
	CCP07 DateTime NULL
)

INSERT INTO @TEMP
(ProjectName,RegistrationNo,EquipmentName,[Description])
SELECT 
DISTINCT A.ProjectName, A.EquipmentName, B.EquipmentName, B.[Description]
FROM [dbo].[tbl_CCPRecord] A
LEFT JOIN [dbo].[tbl_Equipment] B ON A.EquipmentName = B.RegistrationNo
WHERE ActivitiesDate BETWEEN @DateFrom AND @DateTo

UPDATE @TEMP 
SET CCP01 = (SELECT MAX(ApprovedDate) FROM [dbo].[tbl_CCPRecord] WHERE RegistrationNo = EquipmentName AND KeyActivitiesName = 'Entrance Barricade' AND ActivitiesDate BETWEEN @DateFrom AND @DateTo)
UPDATE @TEMP 
SET CCP02 = (SELECT MAX(ApprovedDate) FROM [dbo].[tbl_CCPRecord] WHERE RegistrationNo = EquipmentName AND KeyActivitiesName = 'Life lines & Safety Hook' AND ActivitiesDate BETWEEN @DateFrom AND @DateTo)
UPDATE @TEMP 
SET CCP03 = (SELECT MAX(ApprovedDate) FROM [dbo].[tbl_CCPRecord] WHERE RegistrationNo = EquipmentName AND KeyActivitiesName = 'Hoisting of Guiderails' AND ActivitiesDate BETWEEN @DateFrom AND @DateTo)
UPDATE @TEMP 
SET CCP04 = (SELECT MAX(ApprovedDate) FROM [dbo].[tbl_CCPRecord] WHERE RegistrationNo = EquipmentName AND KeyActivitiesName = 'Setting up of Traction Machine' AND ActivitiesDate BETWEEN @DateFrom AND @DateTo)
UPDATE @TEMP 
SET CCP05 = (SELECT MAX(ApprovedDate) FROM [dbo].[tbl_CCPRecord] WHERE RegistrationNo = EquipmentName AND KeyActivitiesName = 'Hoisting of Control Panel' AND ActivitiesDate BETWEEN @DateFrom AND @DateTo)
UPDATE @TEMP 
SET CCP06 = (SELECT MAX(ApprovedDate) FROM [dbo].[tbl_CCPRecord] WHERE RegistrationNo = EquipmentName AND KeyActivitiesName = 'Hoisting of Cage Platform' AND ActivitiesDate BETWEEN @DateFrom AND @DateTo)
UPDATE @TEMP 
SET CCP07 = (SELECT MAX(ApprovedDate) FROM [dbo].[tbl_CCPRecord] WHERE RegistrationNo = EquipmentName AND KeyActivitiesName = 'Hoisting & securing of Cwt Frame' AND ActivitiesDate BETWEEN @DateFrom AND @DateTo)

SELECT * FROM @TEMP
GO

CREATE PROCEDURE [dbo].[Procedure_GetProjectAllManpowerDetails]
AS

DECLARE @TEMP table 
(
	[Key] [varchar](50),
	[RequestDate] [DateTime],
	[MfgNo] [varchar](2000)
)

INSERT INTO @TEMP
SELECT [Key],CAST(MeetingDate AS DATE) as RequestDate,
	ISNULL(LTRIM(STUFF((SELECT DISTINCT ', ' + CAST(RegistrationNo AS varchar(20)) 
			FROM [dbo].[tbl_TBMEquipments] t2
			WHERE t2.[Key] = t1.[Key] 
			FOR XML PATH('')),1,1,'')),'(No MFG Selected)') as MfgNo
FROM [dbo].[tbl_TBMRecord] t1
WHERE [Status] IN (1,2,3) 
AND ConductedCompany IS NOT NULL

SELECT DISTINCT * FROM
(
SELECT Distinct ProjectName,CAST(MeetingDate AS DATE) as RequestDate,ConductedCompany as RequestCompany,Title + ' ' + LastName + ' ' + FirstName AS RequestBy,Position AS RequestPosition,'AM' TBMSession,MfgNo 
FROM [dbo].[tbl_TBMRecord] Rec
LEFT JOIN [dbo].[tbl_Users] Usr ON Rec.CreatedBy = Usr.UserID
LEFT JOIN @TEMP Tmp ON Rec.[Key] = Tmp.[Key]
WHERE Rec.[Key] IN (SELECT [Key] FROM @TEMP) AND [Description] <> ''
UNION ALL
SELECT Distinct ProjectName,CAST(Tmp.RequestDate AS DATE) as RequestDate,Rec.ConstructorName as RequestCompany,
ISNULL(Title + ' ' + LastName + ' ' + FirstName,Rec.FullName) AS RequestBy,
ISNULL(Usr.Position,Rec.Position) AS RequestPosition,'AM' TBMSession,MfgNo 
FROM [dbo].[tbl_TBMAttendees] Rec
LEFT JOIN [dbo].[tbl_TBMRecord] TBM ON Rec.[Key] = TBM.[Key] 
LEFT JOIN @TEMP Tmp ON Rec.[Key] = Tmp.[Key]
LEFT JOIN [dbo].[tbl_Users] Usr ON Rec.UserID = Usr.UserID
WHERE Tmp.[Key] IS NOT NULL AND [Description] <> ''
UNION ALL
SELECT Distinct ProjectName,CAST(MeetingDate AS DATE) as RequestDate,ConductedCompany as RequestCompany,Title + ' ' + LastName + ' ' + FirstName AS RequestBy,Position AS RequestPosition,'PM' TBMSession,MfgNo 
FROM [dbo].[tbl_TBMRecord] Rec
LEFT JOIN [dbo].[tbl_Users] Usr ON Rec.CreatedBy = Usr.UserID
LEFT JOIN @TEMP Tmp ON Rec.[Key] = Tmp.[Key]
WHERE Rec.[Key] IN (SELECT [Key] FROM @TEMP) AND DescriptionPM <> ''
UNION ALL
SELECT Distinct ProjectName,CAST(Tmp.RequestDate AS DATE) as RequestDate,Rec.ConstructorName as RequestCompany,
ISNULL(Title + ' ' + LastName + ' ' + FirstName,Rec.FullName) AS RequestBy,
ISNULL(Usr.Position,Rec.Position) AS RequestPosition,'PM' TBMSession,MfgNo 
FROM [dbo].[tbl_TBMAttendees] Rec
LEFT JOIN [dbo].[tbl_TBMRecord] TBM ON Rec.[Key] = TBM.[Key] 
LEFT JOIN @TEMP Tmp ON Rec.[Key] = Tmp.[Key]
LEFT JOIN [dbo].[tbl_Users] Usr ON Rec.UserID = Usr.UserID
WHERE Tmp.[Key] IS NOT NULL AND DescriptionPM <> ''
) Details
WHERE RequestCompany IS NOT NULL
ORDER BY RequestDate DESC,ProjectName,RequestCompany

CREATE PROCEDURE [dbo].[Procedure_GetProjectSummary]
	@ProjectName [varchar](150)
AS

SELECT 
'PTW' as Module,
CASE 
WHEN [Status] = 1 THEN 'Submitted'
WHEN [Status] = 2 THEN 'Assessed'
WHEN [Status] = 3 THEN 'Verified'
WHEN [Status] = 4 THEN 'Approved'
WHEN [Status] = 5 THEN 'Closed'
WHEN [Status] = 6 THEN 'Closure Review'
WHEN [Status] = 97 THEN 'Deleted'
WHEN [Status] = 98 THEN 'Returned'
WHEN [Status] = 100 THEN 'Revoked'
ELSE 'Rejected' END as [Status],
Count(*) as [Count]
FROM [dbo].[tbl_PTWRecord]
WHERE ProjectName = @ProjectName
GROUP BY [Status]
UNION ALL
SELECT 
'TBM' as Module,
CASE 
WHEN [Status] = 1 THEN 'Submitted'
WHEN [Status] = 2 THEN 'Approved'
WHEN [Status] = 97 THEN 'Deleted'
WHEN [Status] = 98 THEN 'Returned'
ELSE 'Rejected' END as [Status],
Count(*) as [Count]
FROM [dbo].[tbl_TBMRecord]
WHERE ProjectName = @ProjectName
GROUP BY [Status]
UNION ALL
SELECT 
'CCP' as Module,
CASE 
WHEN [Status] = 1 THEN 'Submitted'
WHEN [Status] = 2 THEN 'Approved'
WHEN [Status] = 97 THEN 'Deleted'
WHEN [Status] = 98 THEN 'Returned'
ELSE 'Rejected' END as [Status],
Count(*) as [Count]
FROM [dbo].[tbl_CCPRecord]
WHERE ProjectName = @ProjectName
GROUP BY [Status]
GO

CREATE PROCEDURE [dbo].[Procedure_GetUserList]
	@UserID [varchar](100)
AS

IF EXISTS(SELECT * FROM [dbo].[tbl_UserRoles] WHERE UserID = @UserID AND RoleID = 'APPLICATION ADMIN')
BEGIN
	SELECT t1.*, 
		LTRIM(STUFF((SELECT DISTINCT ', ' + CAST(RoleID AS varchar(20)) 
			   FROM [dbo].[tbl_UserRoles] t2
			   WHERE t2.UserID = t1.UserID 
			   FOR XML PATH('')),1,1,'')) as Roles
	FROM [dbo].[tbl_Users] t1
	WHERE t1.[Status] = 1
END
ELSE
BEGIN
	IF EXISTS(SELECT * FROM [dbo].[tbl_UserRoles] WHERE UserID = @UserID AND RoleID = 'COMPANY ADMIN')
	BEGIN
		SELECT t1.*, 
			LTRIM(STUFF((SELECT DISTINCT ', ' + CAST(RoleID AS varchar(20)) 
				   FROM [dbo].[tbl_UserRoles] t2
				   WHERE t2.UserID = t1.UserID 
				   FOR XML PATH('')),1,1,'')) as Roles
		FROM [dbo].[tbl_Users] t1
		WHERE t1.ConstructorName IN (SELECT ConstructorName FROM [dbo].[tbl_Users] WHERE UserID = @UserID)
		AND t1.[Status] = 1
	END
	ELSE
	BEGIN
		SELECT t1.*, 
			LTRIM(STUFF((SELECT DISTINCT ', ' + CAST(RoleID AS varchar(20)) 
				   FROM [dbo].[tbl_UserRoles] t2
				   WHERE t2.UserID = t1.UserID 
				   FOR XML PATH('')),1,1,'')) as Roles
		FROM [dbo].[tbl_Users] t1
		WHERE t1.UserID = @UserID
		AND t1.[Status] = 1
	END
END

--DECLARE @ProjectName [varchar](150)
--SET @UserID = 'freddylow.ck.nk@hitachi.com'
--SET @ProjectName = 'Jurong East N2 C26(HDB32)'

DECLARE @TEMP table ([Key] [varchar](50))

DECLARE @MainCont as int
DECLARE @MainContructorName as [varchar](150)
DECLARE @UserContructorName as [varchar](150)

SELECT @MainContructorName = ConstructorName FROM tbl_Projects WHERE [Name] = @ProjectName
SELECT @MainCont = COUNT(*) FROM [dbo].[tbl_Users] WHERE UserID = @UserID AND ConstructorName = @MainContructorName 
SELECT @UserContructorName = ConstructorName FROM [dbo].[tbl_Users] WHERE UserID = @UserID

IF EXISTS(SELECT * FROM [dbo].[tbl_UserRoles] WHERE UserID = @UserID AND RoleID = 'PTW ASSESSOR')
BEGIN

	IF (@MainCont = 1)
	BEGIN
		INSERT INTO @TEMP
		SELECT [Key] FROM [dbo].[tbl_PTWRecord]
		WHERE [Status] = 1 AND [ProjectName] = @ProjectName
	END
	ELSE
	BEGIN
		INSERT INTO @TEMP
		SELECT [Key] FROM [dbo].[tbl_PTWRecord]
		WHERE [Status] = 1 AND [ProjectName] = @ProjectName
		AND RequestCompany = @UserContructorName
	END
END
IF EXISTS(SELECT * FROM [dbo].[tbl_UserRoles] WHERE UserID = @UserID AND RoleID = 'PTW SAFETY')
BEGIN
	INSERT INTO @TEMP
	SELECT [Key] FROM [dbo].[tbl_PTWRecord]
	WHERE [Status] = 2 AND [ProjectName] = @ProjectName
END
IF EXISTS(SELECT * FROM [dbo].[tbl_UserRoles] WHERE UserID = @UserID AND RoleID = 'PTW APPROVER')
BEGIN
	IF (@MainCont = 1)
	BEGIN
		INSERT INTO @TEMP
		SELECT [Key] FROM [dbo].[tbl_PTWRecord]
		WHERE [Status] = 3 AND [ProjectName] = @ProjectName
	END
END

IF EXISTS(SELECT * FROM [dbo].[tbl_UserRoles] WHERE UserID = @UserID AND RoleID = 'PTW ASSESSOR')
BEGIN
	IF (@MainCont = 1)
	BEGIN
		INSERT INTO @TEMP
		SELECT [Key] FROM [dbo].[tbl_PTWRecord]
		WHERE [Status] = 4 AND [ProjectName] = @ProjectName
	END
END
IF EXISTS(SELECT * FROM [dbo].[tbl_UserRoles] WHERE UserID = @UserID AND RoleID = 'PTW SAFETY')
BEGIN
	IF (@MainCont = 1)
	BEGIN
		INSERT INTO @TEMP
		SELECT [Key] FROM [dbo].[tbl_PTWRecord]
		WHERE [Status] = 4 AND [ProjectName] = @ProjectName
	END
END
IF EXISTS(SELECT * FROM [dbo].[tbl_UserRoles] WHERE UserID = @UserID AND RoleID = 'PTW APPROVER')
BEGIN
	IF (@MainCont = 1)
	BEGIN
		INSERT INTO @TEMP
		SELECT [Key] FROM [dbo].[tbl_PTWRecord]
		WHERE [Status] = 4 AND [ProjectName] = @ProjectName
	END
END

INSERT INTO @TEMP
SELECT [Key] FROM [dbo].[tbl_PTWRecord]
WHERE [Status] = 4 AND [ProjectName] = @ProjectName AND RequestBy = @UserID


IF EXISTS(SELECT * FROM [dbo].[tbl_UserRoles] WHERE UserID = @UserID AND RoleID = 'PTW CLOSURE')
BEGIN
	INSERT INTO @TEMP
	SELECT [Key] FROM [dbo].[tbl_PTWRecord]
	WHERE [Status] = 5 AND [ProjectName] = @ProjectName
END

SELECT * FROM [dbo].[tbl_PTWRecord]
WHERE [Key] IN (SELECT [Key] FROM @TEMP)
ORDER BY [STATUS], RequestDate DESC
